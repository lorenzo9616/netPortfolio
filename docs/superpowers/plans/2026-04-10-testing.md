> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a complete test suite across all three services — backend (xUnit), OCR service (pytest), and frontend (Vitest + React Testing Library) — plus Playwright E2E tests and lefthook pre-commit enforcement, all runnable via `make test`.

**Architecture:** Each service has its own independent test suite. Backend tests use EF Core's in-memory database provider (no Docker required). OCR service tests mock Tesseract and pdf2image calls. Frontend tests mock `fetch`. E2E tests run against the live Docker Compose stack. `lefthook` runs all unit suites as a pre-commit hook.

**Tech Stack:** xUnit 2.8 / EF Core InMemory / Microsoft.AspNetCore.Mvc.Testing (backend); pytest / pytest-mock (OCR service); Vitest / @testing-library/react (frontend); Playwright (E2E); lefthook (hook runner)

**Prerequisite:** The DX & Setup plan should be run first — it creates `OcrApi.Tests/`, `Makefile`, and `CONTRIBUTING.md`. If not yet done, Task 1 creates the test project from scratch.

using Microsoft.AspNetCore.Mvc.Testing;
using Microsoft.EntityFrameworkCore;
using Microsoft.Extensions.DependencyInjection;
using OcrApi.Data;

namespace OcrApi.Tests.Helpers;

/// <summary>
/// A <see cref="WebApplicationFactory{TEntryPoint}"/> that replaces the real
/// PostgreSQL database with an in-memory database so tests run without Docker.
/// Also injects test-safe configuration values (JWT secret, dummy credentials).
/// </summary>
public class OcrWebAppFactory : WebApplicationFactory<Program>
{
    private readonly string _dbName = Guid.NewGuid().ToString();

    protected override void ConfigureWebHost(IWebHostBuilder builder)
    {
        builder.ConfigureAppConfiguration((_, cfg) =>
        {
            cfg.AddInMemoryCollection(new Dictionary<string, string?>
            {
                ["Auth:Username"]       = "testuser",
                ["Auth:Password"]       = "testpass",
                ["Auth:JwtSecret"]      = "test-secret-key-that-is-long-enough-32chars",
                ["Auth:JwtExpiryHours"] = "1",
                // Dummy connection string — replaced by InMemory below
                ["ConnectionStrings:DefaultConnection"] = "Host=localhost;Database=test;Username=test;Password=test",
            });
        });

        builder.ConfigureServices(services =>
        {
            // Remove the real Npgsql DbContext registration
            var descriptor = services.SingleOrDefault(
                d => d.ServiceType == typeof(DbContextOptions<OcrDbContext>));
            if (descriptor is not null)
                services.Remove(descriptor);

            // Add InMemory database
            services.AddDbContext<OcrDbContext>(options =>
                options.UseInMemoryDatabase(_dbName));
        });
    }

    /// <summary>
    /// Returns an HttpClient with a valid JWT bearer token for authenticated endpoints.
    /// </summary>
    public async Task<HttpClient> CreateAuthenticatedClientAsync()
    {
        var client = CreateClient();
        var response = await client.PostAsJsonAsync("/auth/login", new
        {
            username = "testuser",
            password = "testpass"
        });
        response.EnsureSuccessStatusCode();
        var body = await response.Content.ReadFromJsonAsync<Dictionary<string, string>>();
        var token = body!["token"];
        client.DefaultRequestHeaders.Authorization =
            new System.Net.Http.Headers.AuthenticationHeaderValue("Bearer", token);
        return client;
    }
}
```

using OcrApi.Models;
using OcrApi.Services;

namespace OcrApi.Tests;

public class FieldMatcherTests
{
    private readonly IFieldMatcher _matcher = new FieldMatcher();

    private static ExtractTextResponse OcrResult(string rawText) => new()
    {
        Success           = true,
        PageCount         = 1,
        RawText           = rawText,
        TextBlocks        = new List<OcrTextBlock>(),
        ProcessingTimeMs  = 0f,
    };

    private static OcrProperty Property(string name, string? heuristic) => new()
    {
        Id              = 1,
        Name            = name,
        DataType        = "string",
        SearchHeuristic = heuristic,
        IsActive        = true,
    };

    [Fact]
    public void MatchFields_PropertyWithMatchingPattern_ReturnsMatch()
    {
        var props  = new List<OcrProperty> { Property("InvoiceNumber", @"INV-\d+") };
        var result = OcrResult("Reference: INV-00423 dated today");

        var fields = _matcher.MatchFields(props, result);

        var field = Assert.Single(fields);
        Assert.Equal("InvoiceNumber", field.PropertyName);
        Assert.Equal("INV-00423", field.ExtractedValue);
    }

    [Fact]
    public void MatchFields_PropertyWithNoMatch_ReturnsNullValue()
    {
        var props  = new List<OcrProperty> { Property("InvoiceNumber", @"INV-\d+") };
        var result = OcrResult("No invoice number here");

        var fields = _matcher.MatchFields(props, result);

        var field = Assert.Single(fields);
        Assert.Equal("InvoiceNumber", field.PropertyName);
        Assert.Null(field.ExtractedValue);
    }

    [Fact]
    public void MatchFields_PropertyWithNoHeuristic_ReturnsNullValue()
    {
        var props  = new List<OcrProperty> { Property("Signature", null) };
        var result = OcrResult("John Smith signed here");

        var fields = _matcher.MatchFields(props, result);

        var field = Assert.Single(fields);
        Assert.Equal("Signature", field.PropertyName);
        Assert.Null(field.ExtractedValue);
    }

    [Fact]
    public void MatchFields_EmptyRawText_ReturnsAllNulls()
    {
        var props  = new List<OcrProperty> { Property("Date", @"\d{2}/\d{2}/\d{4}") };
        var result = OcrResult(string.Empty);

        var fields = _matcher.MatchFields(props, result);

        Assert.Single(fields);
        Assert.Null(fields[0].ExtractedValue);
    }

    [Fact]
    public void MatchFields_MultipleProperties_MatchesEach()
    {
        var props = new List<OcrProperty>
        {
            Property("Date",  @"\d{2}/\d{2}/\d{4}"),
            Property("Total", @"\$[\d,]+\.\d{2}"),
        };
        var result = OcrResult("Date: 03/15/2026 Total: $1,234.56");

        var fields = _matcher.MatchFields(props, result);

        Assert.Equal(2, fields.Count);
        Assert.Equal("03/15/2026", fields.First(f => f.PropertyName == "Date").ExtractedValue);
        Assert.Equal("$1,234.56",  fields.First(f => f.PropertyName == "Total").ExtractedValue);
    }
}

"""Unit tests for extractor.py.

pytesseract.image_to_data is mocked — no real Tesseract installation needed.
"""
import numpy as np
import pandas as pd
import pytest

from extractor import extract_from_image
from models import TextBlock


def _make_tesseract_df(words: list[dict]) -> pd.DataFrame:
    """Build a DataFrame in the format pytesseract.image_to_data returns."""
    defaults = {
        "level": 5, "page_num": 1, "block_num": 1, "par_num": 1,
        "line_num": 1, "word_num": 1, "left": 10, "top": 10,
        "width": 50, "height": 20, "conf": 90, "text": "word",
    }
    rows = [{**defaults, **w} for w in words]
    return pd.DataFrame(rows)


class TestExtractFromImage:
    def test_returns_list_of_text_blocks(self, mocker, blank_image):
        mocker.patch(
            "pytesseract.image_to_data",
            return_value=_make_tesseract_df([{"text": "Hello", "conf": 95}]),
        )
        blocks = extract_from_image(blank_image, page=1)
        assert isinstance(blocks, list)

    def test_each_block_has_text_and_confidence(self, mocker, blank_image):
        mocker.patch(
            "pytesseract.image_to_data",
            return_value=_make_tesseract_df([
                {"text": "Hello", "conf": 95},
                {"text": "World", "conf": 80},
            ]),
        )
        blocks = extract_from_image(blank_image, page=1)
        for block in blocks:
            assert hasattr(block, "text")
            assert hasattr(block, "confidence")
            assert 0 <= block.confidence <= 100

    def test_empty_tesseract_output_returns_empty_list(self, mocker, blank_image):
        mocker.patch(
            "pytesseract.image_to_data",
            return_value=_make_tesseract_df([]),
        )
        blocks = extract_from_image(blank_image, page=1)
        assert blocks == []

    def test_low_confidence_words_filtered(self, mocker, blank_image):
        """Words with conf == -1 (Tesseract non-word rows) should be excluded."""
        mocker.patch(
            "pytesseract.image_to_data",
            return_value=_make_tesseract_df([
                {"text": "GoodWord", "conf": 85},
                {"text": "",          "conf": -1},
            ]),
        )
        blocks = extract_from_image(blank_image, page=1)
        assert all(b.text.strip() != "" for b in blocks)
```
